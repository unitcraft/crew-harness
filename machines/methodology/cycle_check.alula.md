# Машина: Проверка цикла

- machine: cycle_check
- machine_version: alula-2.24
- format: 2.26

Машина «Проверка цикла» методики (задача 122): повтор «проверка → правка» одного артефакта задачи (разбор, план, реализация, сдача) до серии чистых заходов, а для стадии «совещание» вызов вложенной машины совещания. Таблицы состояний — первоисточник методики (23 состояния, 71 строка), записанные в Алуле 2.11; журнальные коды и параметры — таблица журнала методики. Вызывает машину «Раннер» (задача 123) стороной вида `machine cycle_check(вход ВходПроверки, итог ИтогПроверки)`; сама вызывает вложенные машины запуска сессии (`session_start`, задача 121) и совещания (`meeting`, задача 124) и ворота (вратарь, усиленный). Сигналы «стоп», «продолжить», «остановить», «отмена» идут вложенным фрагментом «Пауза».

**Что записано иначе, чем в первоисточнике, и почему** (полный перечень — в разборе задачи). Вердикт — перечисление, а не Текст (предложение методики, «Что движок не покрыл», п. 12). Слова остановки и метка продолжения — наборы литералов, как слова кода автомата. Находки и итоговая строка файла проверки захода приходят в ответе вложенной машины запуска сессии и записываются блоком «При приёме» (правило 14). Режим «показывать и продолжать» у В3 и ответ ворот вратарём-агентом — механика исполнителя ворот, в таблицу не входят.

## Функции

```
функция проверяющий(стадия Стадия) -> строка =>
  выбор стадия { Стадия.Разбор => "s2", Стадия.План => "s4", Стадия.Реализация => "s6", Стадия.Сдача => "s8", иначе => недостижимо }

функция правщик(стадия Стадия) -> строка =>
  выбор стадия { Стадия.Разбор => "s1_fix", Стадия.План => "s3_fix", Стадия.Реализация => "s5_fix", Стадия.Сдача => "s7_fix", иначе => недостижимо }

функция есть_находка_уровня(находки []Находка, уровень Уровень) -> булево =>
  находки.любой(н => н.уровень == уровень)

функция счётчик_доказательства(счётчики []Счётчик, доказательство строка) -> число>=0 {
  пусть найден = счётчики.первый_где(с => с.доказательство == доказательство);
  если есть(найден) то { обязательно(найден).заходов } иначе { 0 }
}

функция обновить_счётчики(счётчики []Счётчик, находки []Находка) -> []Счётчик =>
  находки.свёртка([], (накоплено, н) => если н.доказательство != "" и счётчик_доказательства(накоплено, н.доказательство) == 0 то { накоплено + [Счётчик { доказательство: н.доказательство, заходов: счётчик_доказательства(счётчики, н.доказательство) + 1 }] } иначе { накоплено })

функция наибольший_счётчик(счётчики []Счётчик) -> число>=0 =>
  счётчики.свёртка(0, (самый_большой, с) => наибольшее(самый_большой, с.заходов))

функция сбросить_достигших(счётчики []Счётчик) -> []Счётчик =>
  счётчики.отобразить(с => Счётчик { доказательство: с.доказательство, заходов: если с.заходов >= ПОТОЛОК_ДОКАЗАТЕЛЬСТВА то { 0 } иначе { с.заходов } })

функция без_достигших_потолка(находки []Находка, счётчики []Счётчик) -> []Находка =>
  находки.фильтр(н => счётчик_доказательства(счётчики, н.доказательство) < ПОТОЛОК_ДОКАЗАТЕЛЬСТВА)

функция записи_сужения(находки []Находка, счётчики []Счётчик, что_теряем строка) -> []строка =>
  находки.фильтр(н => счётчик_доказательства(счётчики, н.доказательство) >= ПОТОЛОК_ДОКАЗАТЕЛЬСТВА).отобразить(н => "сужено: " + что_теряем + "; доказательство: " + н.доказательство)

функция находки_цикла(всё словарь[число>=1, []Находка], с_захода число>=1, до_захода число>=1) -> []Находка =>
  (с_захода..до_захода).свёртка([], (накоплено, н) => накоплено + (всё[н] ?? []))
```
